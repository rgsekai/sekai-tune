# The Automix analyser, built for the desktop.
#
# Deliberately the same sources the Android build compiles: `native/analyzer`
# is already platform-neutral C++17 with no dependency beyond the standard
# library, and `app/src/main/cpp/jni` is plain JNI with nothing Android about
# it. Reusing both verbatim is what keeps the desktop's tempo, beat grid and
# mel front end bit-identical to the source of truth rather than a second
# implementation that drifts.
#
# The JNI symbol names encode the Java package, which is why the desktop's
# analysis classes live in `moe.rgsekai.sekaitune.shared.playback.smart` exactly as
# Android's do.

cmake_minimum_required(VERSION 3.16)

if(POLICY CMP0091)
    cmake_policy(SET CMP0091 NEW)
endif()

project(sekai_analysis LANGUAGES CXX)

set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

get_filename_component(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)

foreach(suffix "" _DEBUG _RELEASE _RELWITHDEBINFO _MINSIZEREL)
    set(CMAKE_LIBRARY_OUTPUT_DIRECTORY${suffix} ${CMAKE_BINARY_DIR})
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY${suffix} ${CMAKE_BINARY_DIR})
endforeach()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

add_library(sekai_analysis SHARED
    ${REPO_ROOT}/native/analyzer/audio_analysis.cpp
    ${REPO_ROOT}/native/analyzer/tempo_analysis.cpp
    ${REPO_ROOT}/native/analyzer/resampler.cpp
    ${REPO_ROOT}/native/analyzer/mel_spectrogram.cpp
    ${REPO_ROOT}/native/analyzer/vocal_spectrogram.cpp
    ${REPO_ROOT}/native/jni/analysis_jni.cpp
    ${REPO_ROOT}/native/jni/mel_jni.cpp
    ${REPO_ROOT}/native/jni/vocal_jni.cpp
)

target_include_directories(sekai_analysis PRIVATE
    ${REPO_ROOT}/native
    ${REPO_ROOT}/native/jni
)

# JNI headers come from SEKAI_JDK_HOME or JAVA_HOME
if(DEFINED SEKAI_JDK_HOME)
    target_include_directories(sekai_analysis PRIVATE
        ${SEKAI_JDK_HOME}/include
        ${SEKAI_JDK_HOME}/include/win32
        ${SEKAI_JDK_HOME}/include/linux
        ${SEKAI_JDK_HOME}/include/darwin
    )
elseif(DEFINED JAVA_HOME)
    target_include_directories(sekai_analysis PRIVATE
        ${JAVA_HOME}/include
        ${JAVA_HOME}/include/win32
        ${JAVA_HOME}/include/linux
        ${JAVA_HOME}/include/darwin
    )
elseif(DEFINED ENV{JAVA_HOME})
    target_include_directories(sekai_analysis PRIVATE
        $ENV{JAVA_HOME}/include
        $ENV{JAVA_HOME}/include/linux
        $ENV{JAVA_HOME}/include/win32
        $ENV{JAVA_HOME}/include/darwin
    )
else()
    find_package(JNI REQUIRED)
    target_include_directories(sekai_analysis PRIVATE ${JNI_INCLUDE_DIRS})
endif()

# Cross-building for Windows, the one case where the JDK running the build has
# no header for the platform being built: `jni.h` is shared, `jni_md.h` is not.
# Listed after the JDK's own paths so a real Windows JDK always wins.
if(CMAKE_SYSTEM_NAME STREQUAL "Windows" AND NOT CMAKE_HOST_WIN32)
    target_include_directories(sekai_analysis PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/win32)
endif()

target_compile_options(sekai_analysis PRIVATE -O2)

# `System.mapLibraryName` asks for `sekai_analysis.dll` on Windows, with no
# prefix — but MinGW's CMake defaults to the Unix `lib` one, so a cross-built
# library came out as `libsekai_analysis.dll` and the loader never found it.
# Set on the target rather than in the toolchain file, because `project()`
# reinitialises CMAKE_SHARED_LIBRARY_PREFIX per language and overwrites
# anything the toolchain set beforehand.
if(WIN32)
    set_target_properties(sekai_analysis PROPERTIES PREFIX "")
endif()

# The app unpacks this DLL into its private runtime directory and loads it from
# there, where MinGW's own libstdc++/libgcc/winpthread DLLs are not. Linking them
# statically is what makes the library self-contained; leaving them dynamic makes
# it import libwinpthread-1.dll and fail to load on a machine without MinGW.
if(MINGW)
    target_link_options(sekai_analysis PRIVATE -static -static-libgcc -static-libstdc++)
endif()

# The Windows media controls, which are WinRT. MSVC builds them with C++/WinRT;
# MinGW has no C++/WinRT, so it builds smtc_abi.cpp — the same exports written
# against the raw ABI headers — whenever its headers carry the Media namespace.
include(CheckIncludeFileCXX)
if(MINGW)
    check_include_file_cxx(systemmediatransportcontrolsinterop.h SEKAI_HAVE_SMTC_INTEROP)
    # The same guard smtc_abi.cpp defines: MinGW's windows.foundation.h does not compile without it.
    set(CMAKE_REQUIRED_DEFINITIONS -D____FIReference_1_BYTE_INTERFACE_DEFINED__)
    check_include_file_cxx(windows.media.h SEKAI_HAVE_WINRT_MEDIA)
    unset(CMAKE_REQUIRED_DEFINITIONS)
endif()
if(MINGW AND SEKAI_HAVE_SMTC_INTEROP AND SEKAI_HAVE_WINRT_MEDIA)
    add_library(sekai_smtc SHARED ${CMAKE_CURRENT_SOURCE_DIR}/smtc/smtc_abi.cpp)
    set_target_properties(sekai_smtc PROPERTIES PREFIX "")
    if(DEFINED SEKAI_JDK_HOME)
        target_include_directories(sekai_smtc PRIVATE ${SEKAI_JDK_HOME}/include ${SEKAI_JDK_HOME}/include/win32)
    elseif(DEFINED JAVA_HOME)
        target_include_directories(sekai_smtc PRIVATE ${JAVA_HOME}/include ${JAVA_HOME}/include/win32)
    elseif(DEFINED ENV{JAVA_HOME})
        target_include_directories(sekai_smtc PRIVATE $ENV{JAVA_HOME}/include $ENV{JAVA_HOME}/include/win32)
    else()
        find_package(JNI REQUIRED)
        target_include_directories(sekai_smtc PRIVATE ${JNI_INCLUDE_DIRS})
    endif()
    if(NOT CMAKE_HOST_WIN32)
        target_include_directories(sekai_smtc PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/win32)
    endif()
    target_compile_options(sekai_smtc PRIVATE -O2)
    target_link_libraries(sekai_smtc PRIVATE runtimeobject ole32 user32)
    # Loaded from the app's private runtime directory, where MinGW's own DLLs are not.
    target_link_options(sekai_smtc PRIVATE -static -static-libgcc -static-libstdc++)
elseif(MSVC)
    add_library(sekai_smtc SHARED ${CMAKE_CURRENT_SOURCE_DIR}/smtc/smtc_jni.cpp)
    # C++20 rather than the 17 the analyser uses: under 17, C++/WinRT reaches for
    # <experimental/coroutine>, and current MSVC refuses to compile that header
    # at all. C++20 gives it the standard <coroutine> instead.
    set_target_properties(sekai_smtc PROPERTIES PREFIX "" CXX_STANDARD 20 CXX_STANDARD_REQUIRED ON)
    if(DEFINED SEKAI_JDK_HOME)
        target_include_directories(sekai_smtc PRIVATE ${SEKAI_JDK_HOME}/include ${SEKAI_JDK_HOME}/include/win32)
    elseif(DEFINED JAVA_HOME)
        target_include_directories(sekai_smtc PRIVATE ${JAVA_HOME}/include ${JAVA_HOME}/include/win32)
    elseif(DEFINED ENV{JAVA_HOME})
        target_include_directories(sekai_smtc PRIVATE $ENV{JAVA_HOME}/include $ENV{JAVA_HOME}/include/win32)
    else()
        find_package(JNI REQUIRED)
        target_include_directories(sekai_smtc PRIVATE ${JNI_INCLUDE_DIRS})
    endif()
    # The WinRT umbrella import library.
    target_link_libraries(sekai_smtc PRIVATE windowsapp gdi32)
endif()
